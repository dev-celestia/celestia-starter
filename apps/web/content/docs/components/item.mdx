A flexible list row with media, content, and actions — the building block for settings lists, inboxes, and media feeds.

<ItemPreview />

```tsx
import {
  Item,
  ItemActions,
  ItemContent,
  ItemDescription,
  ItemGroup,
  ItemMedia,
  ItemTitle,
} from "@celestia-project/ui/components/item"

<ItemGroup>
  <Item>
    <ItemMedia>{/* avatar or thumbnail */}</ItemMedia>
    <ItemContent>
      <ItemTitle>@celestia-project/ui</ItemTitle>
      <ItemDescription>Shared component library</ItemDescription>
    </ItemContent>
    <ItemActions>{/* buttons or menu */}</ItemActions>
  </Item>
</ItemGroup>
```
