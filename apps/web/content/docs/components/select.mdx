A styled select popup, built on the Base UI `Select` primitive.

<SelectPreview />

```tsx
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from "@celestia-project/ui/components/select"

<Select defaultValue="apple">
  <SelectTrigger className="w-56">
    <SelectValue placeholder="Select a fruit" />
  </SelectTrigger>
  <SelectContent>
    <SelectItem value="apple">Apple</SelectItem>
    <SelectItem value="banana">Banana</SelectItem>
  </SelectContent>
</Select>
```

Groups and labels are available via `SelectGroup` and `SelectLabel`.
